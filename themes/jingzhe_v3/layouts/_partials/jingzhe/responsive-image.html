{{- $src := strings.TrimSpace (printf "%v" .src) -}} {{- $config := site.Params.services.images | default (dict) -}} {{- $sourceOrigin := replaceRE "/+$" "" (default "" $config.sourceOrigin) -}} {{- $deliveryOrigin := replaceRE "/+$" "" (default $sourceOrigin $config.deliveryOrigin) -}} {{- $thumbWidth := int (default 128 $config.thumbWidth) -}} {{- $smallWidth := int (default 640 $config.smallWidth) -}} {{- $largeWidth := int (default 960 $config.largeWidth) -}} {{- $quality := int (default 75 $config.quality) -}} {{- $thumbSuffix := default "_thumb" $config.thumbnailSuffix -}} {{- $largeSuffix := default "_large" $config.largeSuffix -}} {{- $sourceURL := urls.Parse $src -}} {{- $configuredSourceURL := urls.Parse $sourceOrigin -}} {{- $extension := lower (path.Ext $sourceURL.Path) -}} {{- $isRaster := in (slice ".avif" ".jpg" ".jpeg" ".png" ".webp") $extension -}} {{- $sameOrigin := and (ne $sourceOrigin "") (ne $sourceURL.Host "") (eq (lower $sourceURL.Scheme) (lower $configuredSourceURL.Scheme)) (eq (lower $sourceURL.Host) (lower $configuredSourceURL.Host)) -}} {{- $cleanURL := and (not (in $src "?")) (not (in $src "#")) -}} {{- /* 路线一：Cloudflare 图片缩放（需要 zone 开启 Image Resizing） */ -}} {{- $canTransform := and (default false $config.enabled) (ne $deliveryOrigin "") $sameOrigin $isRaster $cleanURL -}} {{- /* 路线二：按命名约定推导。 编辑器上传文章图时会顺手多传两张：`<原名>_thumb.webp`（640px，卡片用）和 `<原名>_large.webp`（960px，列表大封面用）。这里直接推出地址，前端再用 onerror 兜回原图 —— 老图没有这两张也能正常显示，所以不用迁移任何历史数据。 成本：老图会先发一次 404 再回退，站点图片很少，可以接受。 */ -}} {{- $canDerive := and (default true $config.thumbnailByConvention) (not $canTransform) (ne $sourceOrigin "") $sameOrigin $isRaster $cleanURL (not (strings.Contains $src $thumbSuffix)) (not (strings.Contains $src $largeSuffix)) -}} {{- $small := $src -}} {{- $large := $src -}} {{- $thumb := $src -}} {{- if $canTransform -}} {{- $sourcePath := strings.TrimPrefix "/" (strings.TrimPrefix $sourceOrigin $src) -}} {{- $parameters := "fit=scale-down,format=auto,onerror=redirect" -}} {{- $thumb = printf "%s/cdn-cgi/image/width=%d,quality=%d,%s/%s" $deliveryOrigin $thumbWidth $quality $parameters $sourcePath -}} {{- $small = printf "%s/cdn-cgi/image/width=%d,quality=%d,%s/%s" $deliveryOrigin $smallWidth $quality $parameters $sourcePath -}} {{- $large = printf "%s/cdn-cgi/image/width=%d,quality=%d,%s/%s" $deliveryOrigin $largeWidth $quality $parameters $sourcePath -}} {{- else if $canDerive -}} {{- $directory := path.Dir $sourceURL.Path -}} {{- $baseName := strings.TrimSuffix $extension (path.Base $sourceURL.Path) -}} {{- $prefix := $directory -}} {{- if eq $directory "." -}}{{- $prefix = "" -}}{{- else -}}{{- $prefix = printf "%s/" $directory -}}{{- end -}} {{- $thumb = printf "%s%s%s%s%s" $sourceOrigin $prefix $baseName $thumbSuffix $extension -}} {{- $large = printf "%s%s%s%s%s" $sourceOrigin $prefix $baseName $largeSuffix $extension -}} {{- $small = $thumb -}} {{- end -}} {{- /* disp / dispLarge：两种尺寸槽位各自该用的地址。 没有缩放也没有推导结果时老实回退原图。 */ -}} {{- $disp := $src -}} {{- $dispLarge := $src -}} {{- if or $canTransform $canDerive -}} {{- $disp = $small -}} {{- $dispLarge = $large -}} {{- end -}} {{- return (dict "enabled" $canTransform "derived" $canDerive "original" $src "disp" $disp "dispLarge" $dispLarge "thumb" $thumb "small" $small "large" $large "thumbWidth" $thumbWidth "smallWidth" $smallWidth "largeWidth" $largeWidth ) -}}
